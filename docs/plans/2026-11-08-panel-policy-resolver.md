# Panel Policy — Plan 1: Resolver + Gate Implementation Plan

> **For agentic workers:** REQUIRED: Use `/sh:execute` to implement this plan. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A deterministic `panel_policy.py` resolver (change-type track + census-derived project default + add-only keywords) that `dor_gate.py` consumes, so the DOR gate requires one score line per applicable panel and lets trivial stories skip the panel entirely.

**Architecture:** Stdlib-only Python module vendored byte-identical in `00_Governance/scripts/` and `20_agentflow/scripts/` (the existing `dor_gate.py` pattern), with its data file `panel_policy_data/projects.json` generated by `panel_census.py` and carrying a rows hash. `dor_gate.py` calls `panel_policy.resolve()` once per story, gains a `none` track, and generalises its score-line regex to any `<name>-panel:`. No LLM, no network, no cross-repo import.

**Tech Stack:** Python 3.11+ stdlib (`re`, `json`, `hashlib`, `tomllib`, `subprocess`), pytest. Spec: `docs/superpowers/specs/2026-11-08-panel-policy-design.md` (read it first; this plan does not restate the rationale).

**Out of scope (Plan 2):** lens-pack YAML + loader, roster shrink/deletions, `backlog_dor_pipeline.py` stage E multi-panel, `write_score_line` per-panel filtering, workflow quality-tail text, `/sh:spec-panel` policy header, `dor_score_write.py`.

```python
#!/usr/bin/env python3
"""panel_census.py — per-project signals the panel-policy resolver keys on.

    python3 scripts/panel_census.py [--root ~/projects] [--out scripts/panel_policy_data/projects.json] <project>...

Writes {"generated", "thresholds", "rows_sha256", "rows"}. Rows carry RAW fields only
(the resolver recomputes the rules; see docs/superpowers/specs/2026-11-08-panel-policy-design.md
§Layer 2). rows_sha256 = sha256 of the canonical JSON of rows; panel_policy verifies it so
a hand edit without a re-run is caught. Needs the repos on disk; not run in tests except
over a tmp repo. Stdlib only.
"""
from __future__ import annotations

import argparse
import datetime as dt
import hashlib
import json
import os
import re
import subprocess
import sys
from pathlib import Path

THRESHOLDS = {  # documented here, APPLIED in panel_policy.py — one place each
    "architecture": "(src_dirs>=3 or compose) and (db>=3 or svc>=2), or loc>=15000, or queue>=5",
    "security": "(svc>=2 and cred>=5) or hosts>=10 or cred>=15",
    "test_panel": "test_ratio < 0.20",
    "osm": "geo>=3 and not shapely-only",
}

# ... SRC_EXT, PRUNE, TEST_RE, SIG, HOST_RE, SKIP_HOSTS unchanged from the scratch script ...


def census(p: Path, root: Path) -> dict:
    # body unchanged up to the return; the return becomes:
    return {
        "project": str(p.relative_to(root)),
        "loc": loc, "test_loc": test_loc,
        "test_ratio": round(test_loc / loc, 2) if loc else 0.0,
        "src_dirs": len(top_src_dirs), "compose": compose, "mobile": mobile,
        "db": sig["persist"], "svc": sig["service"], "llm": sig["llm"], "cred": sig["cred"],
        "queue": sig["queue"], "geo": sig["geo"], "hosts": len(hosts),
        "osm": sig["geo"] >= 3 and not _shapely_only(p),
    }


def _shapely_only(p: Path) -> bool:
    """SVG-PAINT's 'geo' hits are shapely geometry, not OpenStreetMap (handover 2026-11-08)."""
    osm_rx = re.compile(r"\b(overpass|taginfo|osm|OpenStreetMap)\b", re.I)
    for f in p.rglob("*.py"):
        if any(seg in PRUNE for seg in f.parts):
            continue
        try:
            if osm_rx.search(f.read_text(encoding="utf-8", errors="ignore")):
                return False
        except OSError:
            continue
    return True


def rows_hash(rows: list[dict]) -> str:
    return hashlib.sha256(json.dumps(rows, sort_keys=True, separators=(",", ":")).encode()).hexdigest()


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    ap.add_argument("--root", type=Path, default=Path(os.path.expanduser("~/projects")))
    ap.add_argument("--out", type=Path, default=Path(__file__).parent / "panel_policy_data" / "projects.json")
    ap.add_argument("projects", nargs="+")
    a = ap.parse_args(argv)
    rows = []
    for name in a.projects:
        p = a.root / name
        if p.is_dir():
            rows.append(census(p, a.root))
    rows.sort(key=lambda r: r["project"])
    doc = {"generated": dt.date.today().isoformat(), "thresholds": THRESHOLDS,
           "rows_sha256": rows_hash(rows), "rows": rows}
    a.out.parent.mkdir(parents=True, exist_ok=True)
    tmp = a.out.with_suffix(".json.tmp")
    tmp.write_text(json.dumps(doc, indent=1) + "\n", encoding="utf-8")
    tmp.replace(a.out)
    print(f"{len(rows)} rows -> {a.out}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
#!/usr/bin/env python3
"""panel_policy.py — deterministic panel resolver for the DOR gate.

    python3 scripts/panel_policy.py resolve <BACKLOG.md> <story-id | title-substring> [--json]

Decides WHICH panels a story must pass (spec: docs/superpowers/specs/2026-11-08-panel-policy-design.md).
Three layers, in order; later layers only add, except track `none` which short-circuits:
  1 change type from the story text and title tags  -> track none | normal | heavy
  2 project default from panel_policy_data/projects.json (census rows; rules recomputed here)
  3 story keywords -> lenses (add-only)
No LLM, no network. Vendored byte-identical in 20_agentflow/scripts and 00_Governance/scripts — edit both.
Stdlib only. Never imports dor_gate (dor_gate imports this).
"""
from __future__ import annotations

import argparse
import hashlib
import json
import os
import re
import subprocess
import sys
import tomllib
from dataclasses import dataclass, field
from pathlib import Path

DATA_DIR = Path(__file__).resolve().parent / "panel_policy_data"
PROJECTS_ENV = "PANEL_POLICY_PROJECTS"  # test/ops override for the projects.json path

# --- layer 1: change type -------------------------------------------------------
AC_RE = re.compile(r"^\s*- (?:\[[ x]\] )?(?:\*\*)?AC-\d+", re.IGNORECASE | re.MULTILINE)  # same shape as dor_gate.AC_RE
TRIVIAL_RE = re.compile(
    r"\b(?:add(?:s|ed|ing)? (?:a |an |the |one )?(?:flag|option|switch)|renam(?:e|es|ing)|default value"
    r"|help text|typo|log line|wording)\b", re.IGNORECASE)
HEAVY_RE = re.compile(
    r"\b(?:new|add(?:s|ed|ing)?|create(?:s|d)?|introduce(?:s|d)?)\b[^.\n]{0,40}\b(?:table|schema|migration|service"
    r"|daemon|integration|queue|scheduled job|cron)\b"
    r"|\b(?:auth|oauth|credential|secret|token)s?\b"
    r"|\bcombines?\b|\bacross\b[^.\n]{0,30}\b(?:capabilit|feature)"
    r"|\b(?:flutter|swift|kotlin|android|ios)\b"
    r"|\bdelet(?:e|es|ing)\b[^.\n]{0,30}\bdata\b", re.IGNORECASE)
TRIVIAL_MAX_AC = 2

# --- layer 3: keywords -> lenses (add-only) --------------------------------------
KEYWORD_LENSES = {
    "business": re.compile(r"\b(?:pricing|launch|market|customer)s?\b", re.IGNORECASE),
}

PANEL_ORDER = ["spec-panel", "architecture-panel", "test-panel"]
LENS_ORDER = ["security", "mobile", "osm", "business"]


@dataclass
class Policy:
    track: str                       # none | normal | heavy
    panels: list[str] = field(default_factory=list)
    lenses: list[str] = field(default_factory=list)
    reasons: list[str] = field(default_factory=list)
    basis: str = "uncensused"        # census:<date> | uncensused | tag

    def as_dict(self) -> dict:
        return {"track": self.track, "panels": self.panels, "lenses": self.lenses,
                "reasons": self.reasons, "basis": self.basis}


def _change_track(text: str, tags: set[str], reasons: list[str]) -> tuple[str, bool]:
    """Returns (track, decided_by_tag)."""
    if "heavy" in tags:
        reasons.append("tag:heavy")
        return "heavy", True
    if "trivial" in tags:
        reasons.append("tag:trivial")
        return "none", True
    m = HEAVY_RE.search(text)
    if m:
        reasons.append(f"heavy:{m.group(0).strip()}")
        return "heavy", False
    m = TRIVIAL_RE.search(text)
    if m:
        n_ac = len(AC_RE.findall(text))
        if n_ac <= TRIVIAL_MAX_AC:
            reasons.append(f"trivial:{m.group(0).strip()} ac={n_ac}")
            return "none", False
        reasons.append(f"trivial-text-but-ac={n_ac}")
    return "normal", False


def resolve(story_text: str, project: str | None, *, title_tags: set[str] | frozenset[str] = frozenset(),
            projects: dict | None = None, repo_root: Path | None = None) -> Policy:
    reasons: list[str] = []
    tags = {t.lower() for t in title_tags}
    track, by_tag = _change_track(story_text, tags, reasons)
    pol = Policy(track=track, reasons=reasons, basis="tag" if by_tag else "uncensused")
    if track == "none":
        return pol
    panels = {"spec-panel"}
    lenses: set[str] = set()
    if track == "heavy":
        panels.add("architecture-panel")
    # layers 2 and 3 are added in Tasks 4 and 5
    pol.panels = [p for p in PANEL_ORDER if p in panels]
    pol.lenses = [l for l in LENS_ORDER if l in lenses]
    return pol
```

```python
class PolicyUnavailable(RuntimeError):
    """projects.json missing, unreadable, or hash-stale. The gate fails closed on this."""


def _rows_hash(rows: list[dict]) -> str:
    return hashlib.sha256(json.dumps(rows, sort_keys=True, separators=(",", ":")).encode()).hexdigest()


def load_projects(path: Path | None = None) -> dict:
    path = path or Path(os.environ.get(PROJECTS_ENV) or DATA_DIR / "projects.json")
    try:
        doc = json.loads(path.read_text(encoding="utf-8"))
    except (OSError, ValueError) as e:
        raise PolicyUnavailable(f"projects.json unreadable: {path}: {e}") from e
    if doc.get("rows_sha256") != _rows_hash(doc.get("rows", [])):
        raise PolicyUnavailable(f"projects.json rows_sha256 mismatch: {path} — re-run scripts/panel_census.py")
    return doc


def project_key(backlog: Path, *, use_git: bool = True) -> str | None:
    """Census key for a BACKLOG: its repo's path relative to ~/projects.
    Worktrees (…/.worktrees/<repo>/<sid>/…) map to <repo>. Outside ~/projects → None (uncensused)."""
    parts = backlog.resolve().parts if backlog.exists() else backlog.parts
    if ".worktrees" in parts:
        return parts[parts.index(".worktrees") + 1]
    top = backlog.parent
    if use_git:
        try:
            out = subprocess.run(["git", "-C", str(backlog.parent), "rev-parse", "--show-toplevel"],
                                 capture_output=True, text=True, check=True, timeout=5).stdout.strip()
            top = Path(out)
        except (subprocess.CalledProcessError, FileNotFoundError, subprocess.TimeoutExpired):
            pass
    try:
        return str(Path(top).resolve().relative_to((Path.home() / "projects").resolve()))
    except ValueError:
        return None
```

```python
# --- CLI -------------------------------------------------------------------------
_HEADING_RE = re.compile(r"^(#{2,4}) (.*)$")
_LIST_TITLE_RE = re.compile(r"^- (?:~~)?\*\*(.+?)\*\*")
_TOP_ITEM_RE = re.compile(r"^(?:- |\d+\. )")
_TAG_RE = re.compile(r"`\[([^\]]+)\]`")
_ID_RE = re.compile(r"\bUS-[A-Z0-9]+(?:-[A-Z0-9]+)*-[0-9]+\b")
_LINK_RE = re.compile(r"\]\(([^)\s]+\.md)\)")


def _find_entry(lines: list[str], selector: str) -> tuple[str, list[str]] | None:
    n = len(lines)
    for i, ln in enumerate(lines):
        h, l = _HEADING_RE.match(ln), _LIST_TITLE_RE.match(ln)
        if not (h or l):
            continue
        title = h.group(2) if h else l.group(1)
        hit = (selector in (m.group(0) for m in _ID_RE.finditer(title))) if _ID_RE.fullmatch(selector) else (selector in title)
        if not hit:
            continue
        j = i + 1
        if h:
            level = len(h.group(1))
            while j < n and not (_HEADING_RE.match(lines[j]) and len(_HEADING_RE.match(lines[j]).group(1)) <= level):
                j += 1
        else:
            while j < n and not (_HEADING_RE.match(lines[j]) or _TOP_ITEM_RE.match(lines[j])):
                j += 1
        return title, lines[i:j]
    return None


def story_input(backlog: Path, selector: str) -> tuple[str, str, set[str]] | None:
    """(story_id, text = entry + linked spec, title tags) for the CLI and for dor_gate."""
    found = _find_entry(backlog.read_text(encoding="utf-8").splitlines(), selector)
    if found is None:
        return None
    title, lines = found
    text = "\n".join(lines)
    link = _LINK_RE.search(text)
    if link:
        spec = (backlog.parent / link.group(1)).resolve()
        if spec.exists():
            text += "\n" + spec.read_text(encoding="utf-8")
    sid = next((m.group(0) for m in _ID_RE.finditer(title)), selector)
    return sid, text, {t.lower() for t in _TAG_RE.findall(lines[0])}


def main(argv: list[str] | None = None) -> int:
    ap = argparse.ArgumentParser(description=__doc__.splitlines()[0])
    sub = ap.add_subparsers(dest="cmd", required=True)
    r = sub.add_parser("resolve", help="print the panel policy for one BACKLOG entry")
    r.add_argument("backlog", type=Path)
    r.add_argument("selector")
    r.add_argument("--json", action="store_true")
    a = ap.parse_args(argv)
    if not a.backlog.exists():
        print(f"not-found: {a.backlog}", file=sys.stderr)
        return 2
    si = story_input(a.backlog, a.selector)
    if si is None:
        print("not-found: entry", file=sys.stderr)
        return 2
    sid, text, tags = si
    key = project_key(a.backlog)
    try:
        pol = resolve(text, key, title_tags=tags, repo_root=a.backlog.parent)
    except PolicyUnavailable as e:
        print(f"policy-unavailable: {e}", file=sys.stderr)
        return 1
    if a.json:
        print(json.dumps({"story": sid, "project": key, **pol.as_dict()}))
    else:
        print(f"story: {sid}\nproject: {key or '(uncensused)'}\ntrack: {pol.track}\n"
              f"panels: {', '.join(pol.panels) or '-'}\nlenses: {', '.join(pol.lenses) or '-'}\n"
              f"basis: {pol.basis}\nreasons: {'; '.join(pol.reasons)}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```
